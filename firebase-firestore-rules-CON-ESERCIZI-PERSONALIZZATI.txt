    }

    match /workoutPlans/{planId} {
      allow read, write: if request.auth != null;
    }

    match /backups/{backupId} {
      allow read, write: if request.auth != null;
    }

    match /workoutTemplates/{templateId} {
      allow read, write: if request.auth != null;
    }

    match /customExercises/{exerciseId} {
      allow read, write: if request.auth != null;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
